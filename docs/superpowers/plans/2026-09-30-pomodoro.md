# Pomodoro com cena ilustrada — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Objetivo:** Web app Pomodoro (foco / pausa curta / pausa longa) exibido dentro de uma cena SVG animada de um ursinho trabalhando num quarto aconchegante.

**Arquitetura:** Lógica pura e testável (`timer.js`, `settings.js`, `sky.js`) separada da apresentação (`scene.js` + CSS) e dos efeitos do navegador (`alerts.js`). `app.js` é o único módulo que conhece todos e faz a ligação. A cena reage a atributos no elemento raiz `#app` (`data-mode`, `data-final`, classes `running` e `celebrating`); todas as animações são CSS.

**Stack:** HTML + CSS + JavaScript puro (ES modules), SVG inline, Web Audio API, Notification API, `localStorage`. Sem dependências e sem build. Testes: mini-runner próprio, executado com Node (`node tests/run.mjs`) e no navegador (`tests/timer.test.html`).

**Spec:** `docs/spec.md`

## Restrições globais

- Sem dependências externas e sem etapa de build; ES modules nativos. Única exceção: fonte Nunito do Google Fonts (com fallback para fonte do sistema).
- Todo texto visível em português do Brasil.
- Arte original — não copiar ilustrações de terceiros.
- Cores por modo: foco `#d8604f` (vermelho), pausa curta `#4f9d69` (verde), pausa longa `#4a6fa5` (azul).
- Padrões e limites: foco 25 (1–120) min, pausa curta 5 (1–60) min, pausa longa 15 (1–60) min, focos até a pausa longa 4 (1–12), iniciar automaticamente = não, som = ligado, volume 70 (0–100), notificação = desligada.
- Chaves de `localStorage`: `pomodoro.settings` e `pomodoro.state`. Toda leitura/escrita dentro de `try/catch`.
- Ao recarregar, o timer volta sempre **pausado**.
- `prefers-reduced-motion: reduce` desliga animações contínuas e piscadas.
- Servidor local: `python -m http.server 5173` na pasta `pomodoro/` (configurado em `.claude/launch.json` como `pomodoro`).
- Commits terminam com a linha `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```js
const tests = [];

export function test(name, fn) {
  tests.push({ name, fn });
}

function canon(value) {
  if (Array.isArray(value)) return value.map(canon);
  if (value && typeof value === 'object') {
    return Object.fromEntries(Object.keys(value).sort().map((k) => [k, canon(value[k])]));
  }
  return value;
}

export function eq(actual, expected, msg = '') {
  const a = JSON.stringify(canon(actual));
  const e = JSON.stringify(canon(expected));
  if (a !== e) throw new Error(`${msg ? `${msg}: ` : ''}esperado ${e}, obtido ${a}`);
}

export async function run() {
  const results = [];
  for (const t of tests) {
    try {
      await t.fn();
      results.push({ name: t.name, ok: true });
    } catch (err) {
      results.push({ name: t.name, ok: false, error: err?.message ?? String(err) });
    }
  }
  return results;
}
```

```js
export const SETTINGS_KEY = 'pomodoro.settings';
export const STATE_KEY = 'pomodoro.state';

export const DEFAULTS = Object.freeze({
  focusMin: 25,
  shortMin: 5,
  longMin: 15,
  cyclesBeforeLong: 4,
  autoStart: false,
  sound: true,
  volume: 70,
  notifications: false,
});

const LIMITS = {
  focusMin: [1, 120],
  shortMin: [1, 60],
  longMin: [1, 60],
  cyclesBeforeLong: [1, 12],
  volume: [0, 100],
};

function toNumber(value) {
  if (typeof value === 'number') return value;
  if (typeof value === 'string' && value.trim() !== '') return Number(value);
  return NaN;
}

export function validate(input) {
  const src = input && typeof input === 'object' ? input : {};
  const out = {};
  for (const [key, fallback] of Object.entries(DEFAULTS)) {
    const value = src[key];
    if (typeof fallback === 'boolean') {
      out[key] = typeof value === 'boolean' ? value : fallback;
      continue;
    }
    const n = toNumber(value);
    if (!Number.isFinite(n)) {
      out[key] = fallback;
      continue;
    }
    const [lo, hi] = LIMITS[key];
    out[key] = Math.min(hi, Math.max(lo, Math.round(n)));
  }
  return out;
}

function defaultStorage() {
  try {
    return globalThis.localStorage ?? null;
  } catch {
    return null;
  }
}

function read(key, storage) {
  try {
    const raw = storage?.getItem(key);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
}

function write(key, value, storage) {
  try {
    storage?.setItem(key, JSON.stringify(value));
  } catch {
    // storage indisponível (modo privado, bloqueado): segue sem salvar
  }
}

export function loadSettings(storage = defaultStorage()) {
  return validate(read(SETTINGS_KEY, storage));
}

export function saveSettings(settings, storage = defaultStorage()) {
  write(SETTINGS_KEY, settings, storage);
}

export function loadState(storage = defaultStorage()) {
  const state = read(STATE_KEY, storage);
  return state && typeof state === 'object' && !Array.isArray(state) ? state : null;
}

export function saveState(state, storage = defaultStorage()) {
  write(STATE_KEY, state, storage);
}
```

```js
export const MODES = Object.freeze(['focus', 'short', 'long']);

const MINUTE = 60_000;

export function durationFor(mode, settings) {
  const minutes = { focus: settings.focusMin, short: settings.shortMin, long: settings.longMin }[mode];
  return minutes * MINUTE;
}

// Reta final: últimos 10 s → 'seconds'; último minuto (ou últimos 20% em blocos < 5 min) → 'minute'.
export function finalStretchLevel(remainingMs, durationMs) {
  if (remainingMs <= 10_000) return 'seconds';
  const threshold = durationMs < 5 * MINUTE ? durationMs * 0.2 : MINUTE;
  return remainingMs <= threshold ? 'minute' : 'none';
}

function restore(saved, settings) {
  const fresh = { mode: 'focus', focusCount: 0, remainingMs: durationFor('focus', settings) };
  if (!saved || typeof saved !== 'object' || !MODES.includes(saved.mode)) return fresh;
  const durationMs = durationFor(saved.mode, settings);
  const focusCount = Number.isInteger(saved.focusCount)
    ? Math.min(Math.max(saved.focusCount, 0), settings.cyclesBeforeLong)
    : 0;
  const remainingMs = Number.isFinite(saved.remainingMs) && saved.remainingMs > 0
    ? Math.min(saved.remainingMs, durationMs)
    : durationMs;
  return { mode: saved.mode, focusCount, remainingMs };
}

export function createTimer(initialSettings, saved = null, now = () => Date.now()) {
  let settings = initialSettings;
  let { mode, focusCount, remainingMs } = restore(saved, settings);
  let durationMs = durationFor(mode, settings);
  let endsAt = null; // instante de término enquanto roda; null quando pausado

  const running = () => endsAt !== null;
  const remaining = () => (running() ? Math.max(0, endsAt - now()) : remainingMs);

  function advance() {
    if (mode === 'focus') {
      focusCount += 1;
      mode = focusCount >= settings.cyclesBeforeLong ? 'long' : 'short';
    } else {
      if (mode === 'long') focusCount = 0;
      mode = 'focus';
    }
    durationMs = durationFor(mode, settings);
    remainingMs = durationMs;
    endsAt = settings.autoStart ? now() + durationMs : null;
  }

  return {
    start() {
      if (!running()) endsAt = now() + remainingMs;
    },
    pause() {
      if (!running()) return;
      remainingMs = remaining();
      endsAt = null;
    },
    reset() {
      endsAt = null;
      remainingMs = durationMs;
    },
    skip() {
      advance();
    },
    tick() {
      if (!running() || remaining() > 0) return false;
      advance();
      return true;
    },
    getState() {
      return { mode, focusCount, remainingMs: remaining(), durationMs, running: running() };
    },
    updateSettings(next) {
      settings = next;
      if (!running() && remainingMs === durationMs) {
        durationMs = durationFor(mode, settings);
        remainingMs = durationMs;
      }
      focusCount = Math.min(focusCount, settings.cyclesBeforeLong);
    },
  };
}
```

```js
// Pontos do céu ao longo do bloco: dia → pôr do sol → noite.
const SKY = [
  { at: 0, top: [126, 200, 240], bottom: [205, 238, 253] },
  { at: 0.6, top: [240, 138, 93], bottom: [255, 210, 138] },
  { at: 0.9, top: [28, 35, 64], bottom: [59, 63, 115] },
  { at: 1, top: [28, 35, 64], bottom: [59, 63, 115] },
];

const clamp01 = (x) => Math.min(1, Math.max(0, x));

function mix(from, to, t) {
  return `#${from.map((v, i) => Math.round(v + (to[i] - v) * t).toString(16).padStart(2, '0')).join('')}`;
}

export function skyAt(progress) {
  const p = clamp01(Number.isFinite(progress) ? progress : 0);
  let i = 0;
  while (i < SKY.length - 2 && p > SKY[i + 1].at) i++;
  const a = SKY[i];
  const b = SKY[i + 1];
  const t = (p - a.at) / (b.at - a.at);
  return {
    top: mix(a.top, b.top, t),
    bottom: mix(a.bottom, b.bottom, t),
    sunY: 110 + 150 * Math.min(1, p / 0.7),
    night: clamp01((p - 0.7) / 0.2),
  };
}
```

```js
import { skyAt } from './sky.js';

const SVG_NS = 'http://www.w3.org/2000/svg';
const CONFETTI_COLORS = ['#ffd166', '#ef476f', '#06d6a0', '#118ab2', '#fff8ee'];
const STAR_PATH = 'M0-9L2.5-2.5L9 0L2.5 2.5L0 9L-2.5 2.5L-9 0L-2.5-2.5Z';

const rand = (min, max) => min + Math.random() * (max - min);

function prefersReducedMotion() {
  return window.matchMedia?.('(prefers-reduced-motion: reduce)').matches ?? false;
}

function spawnConfetti(layer) {
  const pieces = [];
  for (let i = 0; i < 40; i++) {
    const star = i % 3 === 0;
    const g = document.createElementNS(SVG_NS, 'g');
    g.setAttribute('transform', `translate(${rand(40, 960).toFixed(0)} ${rand(-60, -10).toFixed(0)})`);
    const piece = document.createElementNS(SVG_NS, star ? 'path' : 'rect');
    if (star) {
      piece.setAttribute('d', STAR_PATH);
    } else {
      piece.setAttribute('x', '-4');
      piece.setAttribute('y', '-6');
      piece.setAttribute('width', '8');
      piece.setAttribute('height', '12');
      piece.setAttribute('rx', '2');
    }
    piece.setAttribute('class', 'confetti-piece');
    piece.setAttribute('fill', star ? '#ffd166' : CONFETTI_COLORS[i % CONFETTI_COLORS.length]);
    piece.style.setProperty('--dx', `${rand(-90, 90).toFixed(0)}px`);
    piece.style.setProperty('--rot', `${rand(300, 900).toFixed(0)}deg`);
    piece.style.setProperty('--dur', `${rand(1, 1.5).toFixed(2)}s`);
    piece.style.setProperty('--delay', `${rand(0, 0.3).toFixed(2)}s`);
    g.append(piece);
    pieces.push(g);
  }
  layer.replaceChildren(...pieces);
}

export function createScene(root) {
  const skyTop = root.querySelector('#sky-top');
  const skyBottom = root.querySelector('#sky-bottom');
  const sun = root.querySelector('#sun');
  const nightLayers = [root.querySelector('#stars'), root.querySelector('#moon')];
  const confetti = root.querySelector('#confetti');
  let lastProgress = -1;

  return {
    setMode(mode) {
      root.dataset.mode = mode;
    },
    setRunning(running) {
      root.classList.toggle('running', running);
    },
    setProgress(fraction) {
      if (Math.abs(fraction - lastProgress) < 0.001) return;
      lastProgress = fraction;
      const sky = skyAt(fraction);
      skyTop.setAttribute('stop-color', sky.top);
      skyBottom.setAttribute('stop-color', sky.bottom);
      sun.setAttribute('cy', sky.sunY.toFixed(1));
      for (const layer of nightLayers) layer.setAttribute('opacity', sky.night.toFixed(2));
    },
    setFinalStretch(level) {
      root.dataset.final = level;
    },
    celebrate() {
      const reduced = prefersReducedMotion();
      root.classList.add('celebrating');
      if (!reduced) spawnConfetti(confetti);
      return new Promise((resolve) => {
        setTimeout(() => {
          root.classList.remove('celebrating');
          confetti.replaceChildren();
          resolve();
        }, reduced ? 800 : 2000);
      });
    },
  };
}
```

```js
let ctx = null;

// Navegadores só liberam áudio após um gesto do usuário: chamar em cliques.
export function unlockAudio() {
  try {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    if (!AudioCtx) return;
    if (!ctx) ctx = new AudioCtx();
    if (ctx.state === 'suspended') ctx.resume();
  } catch {
    // sem áudio disponível
  }
}

// Fim do foco: melodia subindo. Fim da pausa: descendo.
const MELODIES = {
  focus: [[784, 0], [988, 0.18], [1175, 0.36]],
  break: [[1175, 0], [988, 0.18], [784, 0.36]],
};

export function playEnd(endedMode, volume) {
  try {
    if (!ctx || volume <= 0) return;
    const notes = endedMode === 'focus' ? MELODIES.focus : MELODIES.break;
    const t0 = ctx.currentTime + 0.05;
    const peak = (volume / 100) * 0.3;
    for (const [freq, offset] of notes) {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'triangle';
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0.0001, t0 + offset);
      gain.gain.exponentialRampToValueAtTime(peak, t0 + offset + 0.02);
      gain.gain.exponentialRampToValueAtTime(0.0001, t0 + offset + 0.35);
      osc.connect(gain).connect(ctx.destination);
      osc.start(t0 + offset);
      osc.stop(t0 + offset + 0.4);
    }
  } catch {
    // falha de áudio não interrompe o timer
  }
}

export async function requestPermission() {
  try {
    if (!('Notification' in window)) return false;
    if (Notification.permission === 'granted') return true;
    if (Notification.permission === 'denied') return false;
    return (await Notification.requestPermission()) === 'granted';
  } catch {
    return false;
  }
}

export function notify(title, body) {
  try {
    if ('Notification' in window && Notification.permission === 'granted') {
      new Notification(title, { body });
    }
  } catch {
    // notificação indisponível
  }
}
```

```js
    els.cycles.innerHTML = Array.from({ length: total }, () => `<span class="cycle">${CHEVRON}</span>`).join('');
  }
  [...els.cycles.children].forEach((el, i) => el.classList.toggle('done', i < focusCount));
  els.cycles.setAttribute('aria-label', `${Math.min(focusCount, total)} de ${total} focos concluídos`);
}

function render() {
  const st = timer.getState();
  const time = format(st.remainingMs);
  els.time.textContent = time;
  document.title = `${EMOJI[st.mode]} ${time} – ${LABELS[st.mode]}`;
  els.toggle.dataset.state = st.running ? 'running' : 'paused';
  els.toggle.setAttribute('aria-label', st.running ? 'Pausar' : 'Iniciar');
  renderCycles(st.focusCount);
  if (celebrating) return; // a cena termina a comemoração no modo antigo
  scene.setMode(st.mode);
  scene.setRunning(st.running);
  scene.setProgress(1 - st.remainingMs / st.durationMs);
  scene.setFinalStretch(finalStretchLevel(st.remainingMs, st.durationMs));
}

async function onBlockEnd(endedMode) {
  persist();
  if (settings.sound) playEnd(endedMode, settings.volume);
  if (settings.notifications) {
    notify(endedMode === 'focus' ? 'Foco concluído!' : 'Pausa terminada!', activity || 'Hora do próximo bloco.');
  }
  celebrating = true;
  scene.setFinalStretch('none');
  await scene.celebrate();
  celebrating = false;
  render();
}

function onTick() {
  const endedMode = timer.getState().mode;
  if (timer.tick()) onBlockEnd(endedMode);
  render();
  if (timer.getState().running && Date.now() - lastSaveAt > 5000) persist();
}

// Controles do timer
els.toggle.addEventListener('click', () => {
  unlockAudio();
  if (timer.getState().running) timer.pause();
  else timer.start();
  persist();
  render();
});

els.reset.addEventListener('click', () => {
  const st = timer.getState();
  if (st.remainingMs < st.durationMs && !confirm('Reiniciar o bloco atual?')) return;
  timer.reset();
  persist();
  render();
});

els.skip.addEventListener('click', () => {
  timer.skip();
  persist();
  render();
});

// Atividade
els.activity.value = activity;
els.activity.addEventListener('input', () => {
  activity = els.activity.value.slice(0, 80);
  persist();
});
els.activity.addEventListener('keydown', (e) => {
  if (e.key === 'Enter') els.activity.blur();
});

// Configurações
function fillForm(s) {
  const f = els.form.elements;
  for (const key of NUMBER_FIELDS) f[key].value = s[key];
  for (const key of BOOL_FIELDS) f[key].checked = s[key];
  els.notifHint.hidden = true;
}

function readForm() {
  const f = els.form.elements;
  const values = {};
  for (const key of NUMBER_FIELDS) values[key] = f[key].value;
  for (const key of BOOL_FIELDS) values[key] = f[key].checked;
  return validate(values);
}

els.openSettings.addEventListener('click', () => {
  fillForm(settings);
  els.dialog.returnValue = '';
  els.dialog.showModal();
});

els.form.elements.notifications.addEventListener('change', async (e) => {
  if (!e.target.checked) return;
  const granted = await requestPermission();
  if (!granted) {
    e.target.checked = false;
    els.notifHint.hidden = false;
  }
});

els.dialog.addEventListener('close', () => {
  if (els.dialog.returnValue !== 'save') return;
  unlockAudio();
  settings = readForm();
  saveSettings(settings);
  timer.updateSettings(settings);
  persist();
  render();
});

window.addEventListener('pagehide', persist);

render();
setInterval(onTick, 250);
```
